CREATE OR REPLACE FUNCTION public.handle_new_bodyland_user() RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ BEGIN INSERT INTO public.athletes ( user_id, full_name, phone, username, email, membership_status, total_sessions, remaining_sessions, coaching_type ) VALUES ( NEW.id, COALESCE( NEW.raw_user_meta_data->>'full_name', '' ), NEW.raw_user_meta_data->>'phone', LOWER(NEW.raw_user_meta_data->>'username'), NEW.raw_user_meta_data->>'contact_email', 'pending', 0, 0, 'none' ); RETURN NEW; END; $$; DROP TRIGGER IF EXISTS on_bodyland_user_created ON auth.users; CREATE TRIGGER on_bodyland_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION public.handle_new_bodyland_user();
